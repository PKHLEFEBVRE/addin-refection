Option Explicit

' Generates a compliance report for a given portfolio
' Evaluates rules dynamically based on what is populated in port.exposureLimits
Public Function GenerateComplianceReport(port As Portfolio) As Variant

    Dim reportRows As New Collection
    Dim headerRow(1 To 6) As Variant

    ' Headers
    headerRow(1) = "Rule Name"
    headerRow(2) = "Type"
    headerRow(3) = "Current Exposure"
    headerRow(4) = "Warning Level"
    headerRow(5) = "Breach Level"
    headerRow(6) = "Status"
    reportRows.Add headerRow

    Dim exp As Double
    Dim limitBreach As Double
    Dim res As Variant
    Dim ruleKey As Variant
    Dim rowData(1 To 6) As Variant

    ' If no limits defined, return just the headers
    If port.exposureLimits Is Nothing Then
        GenerateComplianceReport = CollectionTo2DArray(reportRows)
        Exit Function
    End If
    If port.exposureLimits.count = 0 Then
        GenerateComplianceReport = CollectionTo2DArray(reportRows)
        Exit Function
    End If

    ' Cache some calculations that might be used by multiple rules
    Dim deltaEquity As Double: deltaEquity = -999
    Dim bondsExp As Double: bondsExp = -999

    For Each ruleKey In port.exposureLimits.Keys
        limitBreach = port.exposureLimits(ruleKey)

        Select Case UCase(Trim(ruleKey))
            Case "DELTAEQUITY_MIN"
                If deltaEquity = -999 Then deltaEquity = port.GetEquityDelta()
                res = EvaluateThreshold(deltaEquity, limitBreach, "MIN")
                reportRows.Add CreateRow("Delta Equity Minimum", "MIN", deltaEquity, res(0), limitBreach, res(1))

            Case "DELTAEQUITY_MAX"
                If deltaEquity = -999 Then deltaEquity = port.GetEquityDelta()
                res = EvaluateThreshold(deltaEquity, limitBreach, "MAX")
                reportRows.Add CreateRow("Delta Equity Maximum", "MAX", deltaEquity, res(0), limitBreach, res(1))

            Case "BONDS_MIN"
                If bondsExp = -999 Then bondsExp = Calc_Bonds(port)
                res = EvaluateThreshold(bondsExp, limitBreach, "MIN")
                reportRows.Add CreateRow("Bonds Minimum", "MIN", bondsExp, res(0), limitBreach, res(1))

            Case "BONDS_MAX"
                If bondsExp = -999 Then bondsExp = Calc_Bonds(port)
                res = EvaluateThreshold(bondsExp, limitBreach, "MAX")
                reportRows.Add CreateRow("Bonds Maximum", "MAX", bondsExp, res(0), limitBreach, res(1))

            Case "ALTERNATIVE_MAX"
                exp = Calc_Alternative(port)
                res = EvaluateThreshold(exp, limitBreach, "MAX")
                reportRows.Add CreateRow("Alternative Maximum", "MAX", exp, res(0), limitBreach, res(1))

            Case "RISKOFF_MIN"
                exp = Calc_RiskOff(port)
                res = EvaluateThreshold(exp, limitBreach, "MIN")
                reportRows.Add CreateRow("Risk Off Minimum", "MIN", exp, res(0), limitBreach, res(1))

            Case "STRUCTURED_MAX"
                exp = Calc_StructuredProducts(port)
                res = EvaluateThreshold(exp, limitBreach, "MAX")
                reportRows.Add CreateRow("Structured Products Maximum", "MAX", exp, res(0), limitBreach, res(1))

            ' You can easily add more cases here for new rules
        End Select
    Next ruleKey

    GenerateComplianceReport = CollectionTo2DArray(reportRows)

End Function

Private Function CreateRow(v1 As Variant, v2 As Variant, v3 As Variant, v4 As Variant, v5 As Variant, v6 As Variant) As Variant
    Dim arr(1 To 6) As Variant
    arr(1) = v1
    arr(2) = v2
    arr(3) = v3
    arr(4) = v4
    arr(5) = v5
    arr(6) = v6
    CreateRow = arr
End Function

' Converts a Collection of 1D arrays into a 2D Variant array
Private Function CollectionTo2DArray(col As Collection) As Variant
    If col.count = 0 Then
        CollectionTo2DArray = Array()
        Exit Function
    End If

    Dim r As Long, c As Long
    Dim numRows As Long, numCols As Long
    numRows = col.count
    numCols = UBound(col.Item(1)) ' Assumes all items are 1-based arrays of same length

    Dim result() As Variant
    ReDim result(1 To numRows, 1 To numCols)

    Dim rowArr As Variant
    For r = 1 To numRows
        rowArr = col.Item(r)
        For c = 1 To numCols
            result(r, c) = rowArr(c) ' Assuming rowArr is 1-based from Array() function or explicit dim
        Next c
    Next r

    CollectionTo2DArray = result
End Function

' Evaluates exposure against a breach limit and returns an Array(WarningLevel, Status)
Private Function EvaluateThreshold(exposure As Double, breachLimit As Double, ruleType As String) As Variant
    Dim warningLimit As Double
    Dim status As String

    If ruleType = "MAX" Then
        warningLimit = breachLimit * 0.9
        If exposure > breachLimit Then
            status = "BREACH"
        ElseIf exposure > warningLimit Then
            status = "WARNING"
        Else
            status = "PASS"
        End If
    ElseIf ruleType = "MIN" Then
        warningLimit = breachLimit * 1.1
        If exposure < breachLimit Then
            status = "BREACH"
        ElseIf exposure < warningLimit Then
            status = "WARNING"
        Else
            status = "PASS"
        End If
    Else
        status = "UNKNOWN"
        warningLimit = 0
    End If

    ' Return 1-based array to match the CollectionTo2DArray expectation if needed,
    ' or just return standard 0-based Array. Here using 0-based for internal use.
    EvaluateThreshold = Array(warningLimit, status)
End Function

' ---------------------------------------------------------
' Rule Calculation Helpers
' ---------------------------------------------------------

Private Function Calc_Bonds(port As Portfolio) As Double
    Dim sum As Double
    Dim key As Variant
    Dim pos As Position
    Dim inst As Instrument

    If port.positions Is Nothing Then Exit Function

    For Each key In port.positions
        Set pos = port.positions(key)
        Set inst = pos.posInstrument

        If Not inst Is Nothing Then
            If LCase(Trim(inst.assetClass)) = "bond" And LCase(Trim(inst.category)) <> "certificates" Then
                sum = sum + pos.weight
            End If
        End If
    Next key

    Calc_Bonds = sum
End Function

Private Function Calc_Alternative(port As Portfolio) As Double
    Dim sum As Double
    Dim key As Variant
    Dim pos As Position
    Dim inst As Instrument

    If port.positions Is Nothing Then Exit Function

    For Each key In port.positions
        Set pos = port.positions(key)
        Set inst = pos.posInstrument

        If Not inst Is Nothing Then
            If LCase(Trim(inst.category)) = "alternatives investment" Then
                sum = sum + pos.weight
            End If
        End If
    Next key

    Calc_Alternative = sum
End Function

Private Function Calc_RiskOff(port As Portfolio) As Double
    Dim sum As Double
    Dim key As Variant
    Dim pos As Position
    Dim inst As Instrument
    Dim isBond As Boolean

    If port.positions Is Nothing Then Exit Function

    For Each key In port.positions
        Set pos = port.positions(key)
        Set inst = pos.posInstrument

        If Not inst Is Nothing Then
            isBond = (LCase(Trim(inst.assetClass)) = "bond" And LCase(Trim(inst.category)) <> "certificates")

            If isBond Or LCase(Trim(inst.assetClass)) = "real estate" Or LCase(Trim(inst.assetClass)) = "cash" Then
                sum = sum + pos.weight
            End If
        End If
    Next key

    Calc_RiskOff = sum
End Function

Private Function Calc_StructuredProducts(port As Portfolio) As Double
    Dim sum As Double
    Dim key As Variant
    Dim pos As Position
    Dim inst As Instrument

    If port.positions Is Nothing Then Exit Function

    For Each key In port.positions
        Set pos = port.positions(key)
        Set inst = pos.posInstrument

        If Not inst Is Nothing Then
            If LCase(Trim(inst.category)) = "certificates" Then
                sum = sum + pos.weight
            End If
        End If
    Next key

    Calc_StructuredProducts = sum
End Function
