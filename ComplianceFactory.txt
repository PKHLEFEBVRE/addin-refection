Option Explicit

' Generates a compliance report for a given portfolio
' Evaluates hardcoded rules against the portfolio's positions
Public Function GenerateComplianceReport(port As Portfolio) As Variant

    Dim rulesCount As Integer
    rulesCount = 7 ' Delta Min, Delta Max, Bonds Min, Bonds Max, Alternative Max, Risk Off Min, Structured Max

    Dim report() As Variant
    ReDim report(1 To rulesCount + 1, 1 To 6)

    ' Headers
    report(1, 1) = "Rule Name"
    report(1, 2) = "Type"
    report(1, 3) = "Current Exposure"
    report(1, 4) = "Warning Level"
    report(1, 5) = "Breach Level"
    report(1, 6) = "Status"

    Dim r As Integer
    r = 2

    Dim exp As Double
    Dim limitBreach As Double
    Dim res As Variant

    ' --- Rule 1: Delta Equity Minimum ---
    exp = port.GetEquityDelta()
    If port.exposureLimits.Exists("DeltaEquity_Min") Then limitBreach = port.exposureLimits("DeltaEquity_Min") Else limitBreach = 0
    res = EvaluateThreshold(exp, limitBreach, "MIN")
    report(r, 1) = "Delta Equity Minimum"
    report(r, 2) = "MIN"
    report(r, 3) = exp
    report(r, 4) = res(0) ' Warning Level
    report(r, 5) = limitBreach ' Breach Level
    report(r, 6) = res(1) ' Status
    r = r + 1

    ' --- Rule 2: Delta Equity Maximum ---
    ' Re-use exp
    If port.exposureLimits.Exists("DeltaEquity_Max") Then limitBreach = port.exposureLimits("DeltaEquity_Max") Else limitBreach = 1
    res = EvaluateThreshold(exp, limitBreach, "MAX")
    report(r, 1) = "Delta Equity Maximum"
    report(r, 2) = "MAX"
    report(r, 3) = exp
    report(r, 4) = res(0)
    report(r, 5) = limitBreach
    report(r, 6) = res(1)
    r = r + 1

    ' --- Rule 3: Bonds Minimum ---
    exp = Calc_Bonds(port)
    If port.exposureLimits.Exists("Bonds_Min") Then limitBreach = port.exposureLimits("Bonds_Min") Else limitBreach = 0
    res = EvaluateThreshold(exp, limitBreach, "MIN")
    report(r, 1) = "Bonds Minimum"
    report(r, 2) = "MIN"
    report(r, 3) = exp
    report(r, 4) = res(0)
    report(r, 5) = limitBreach
    report(r, 6) = res(1)
    r = r + 1

    ' --- Rule 4: Bonds Maximum ---
    ' Re-use exp
    If port.exposureLimits.Exists("Bonds_Max") Then limitBreach = port.exposureLimits("Bonds_Max") Else limitBreach = 1
    res = EvaluateThreshold(exp, limitBreach, "MAX")
    report(r, 1) = "Bonds Maximum"
    report(r, 2) = "MAX"
    report(r, 3) = exp
    report(r, 4) = res(0)
    report(r, 5) = limitBreach
    report(r, 6) = res(1)
    r = r + 1

    ' --- Rule 5: Alternative Maximum ---
    exp = Calc_Alternative(port)
    If port.exposureLimits.Exists("Alternative_Max") Then limitBreach = port.exposureLimits("Alternative_Max") Else limitBreach = 0.1
    res = EvaluateThreshold(exp, limitBreach, "MAX")
    report(r, 1) = "Alternative Maximum"
    report(r, 2) = "MAX"
    report(r, 3) = exp
    report(r, 4) = res(0)
    report(r, 5) = limitBreach
    report(r, 6) = res(1)
    r = r + 1

    ' --- Rule 6: Risk Off Minimum ---
    exp = Calc_RiskOff(port)
    If port.exposureLimits.Exists("RiskOff_Min") Then limitBreach = port.exposureLimits("RiskOff_Min") Else limitBreach = 0
    res = EvaluateThreshold(exp, limitBreach, "MIN")
    report(r, 1) = "Risk Off Minimum"
    report(r, 2) = "MIN"
    report(r, 3) = exp
    report(r, 4) = res(0)
    report(r, 5) = limitBreach
    report(r, 6) = res(1)
    r = r + 1

    ' --- Rule 7: Structured Products Maximum ---
    exp = Calc_StructuredProducts(port)
    If port.exposureLimits.Exists("Structured_Max") Then limitBreach = port.exposureLimits("Structured_Max") Else limitBreach = 0.1
    res = EvaluateThreshold(exp, limitBreach, "MAX")
    report(r, 1) = "Structured Products Maximum"
    report(r, 2) = "MAX"
    report(r, 3) = exp
    report(r, 4) = res(0)
    report(r, 5) = limitBreach
    report(r, 6) = res(1)

    GenerateComplianceReport = report

End Function

' Evaluates exposure against a breach limit and returns an Array(WarningLevel, Status)
Private Function EvaluateThreshold(exposure As Double, breachLimit As Double, ruleType As String) As Variant
    Dim warningLimit As Double
    Dim status As String

    If ruleType = "MAX" Then
        warningLimit = breachLimit * 0.9
        If exposure > breachLimit Then
            status = "BREACH"
        ElseIf exposure > warningLimit Then
            status = "WARNING"
        Else
            status = "PASS"
        End If
    ElseIf ruleType = "MIN" Then
        warningLimit = breachLimit * 1.1
        If exposure < breachLimit Then
            status = "BREACH"
        ElseIf exposure < warningLimit Then
            status = "WARNING"
        Else
            status = "PASS"
        End If
    Else
        status = "UNKNOWN"
        warningLimit = 0
    End If

    EvaluateThreshold = Array(warningLimit, status)
End Function

' ---------------------------------------------------------
' Rule Calculation Helpers
' ---------------------------------------------------------

Private Function Calc_Bonds(port As Portfolio) As Double
    Dim sum As Double
    Dim key As Variant
    Dim pos As Position
    Dim inst As Instrument

    If port.positions Is Nothing Then Exit Function

    For Each key In port.positions
        Set pos = port.positions(key)
        Set inst = pos.posInstrument

        If Not inst Is Nothing Then
            If LCase(Trim(inst.assetClass)) = "bond" And LCase(Trim(inst.category)) <> "certificates" Then
                sum = sum + pos.weight
            End If
        End If
    Next key

    Calc_Bonds = sum
End Function

Private Function Calc_Alternative(port As Portfolio) As Double
    Dim sum As Double
    Dim key As Variant
    Dim pos As Position
    Dim inst As Instrument

    If port.positions Is Nothing Then Exit Function

    For Each key In port.positions
        Set pos = port.positions(key)
        Set inst = pos.posInstrument

        If Not inst Is Nothing Then
            If LCase(Trim(inst.category)) = "alternatives investment" Then
                sum = sum + pos.weight
            End If
        End If
    Next key

    Calc_Alternative = sum
End Function

Private Function Calc_RiskOff(port As Portfolio) As Double
    Dim sum As Double
    Dim key As Variant
    Dim pos As Position
    Dim inst As Instrument
    Dim isBond As Boolean

    If port.positions Is Nothing Then Exit Function

    For Each key In port.positions
        Set pos = port.positions(key)
        Set inst = pos.posInstrument

        If Not inst Is Nothing Then
            isBond = (LCase(Trim(inst.assetClass)) = "bond" And LCase(Trim(inst.category)) <> "certificates")

            If isBond Or LCase(Trim(inst.assetClass)) = "real estate" Or LCase(Trim(inst.assetClass)) = "cash" Then
                sum = sum + pos.weight
            End If
        End If
    Next key

    Calc_RiskOff = sum
End Function

Private Function Calc_StructuredProducts(port As Portfolio) As Double
    Dim sum As Double
    Dim key As Variant
    Dim pos As Position
    Dim inst As Instrument

    If port.positions Is Nothing Then Exit Function

    For Each key In port.positions
        Set pos = port.positions(key)
        Set inst = pos.posInstrument

        If Not inst Is Nothing Then
            If LCase(Trim(inst.category)) = "certificates" Then
                sum = sum + pos.weight
            End If
        End If
    Next key

    Calc_StructuredProducts = sum
End Function
