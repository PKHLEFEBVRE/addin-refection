Option Explicit

' Generates a compliance report for a given portfolio
' Evaluates rules dynamically based on what is populated in port.exposureLimits
Public Function GenerateComplianceReport(port As Portfolio) As Variant

    Dim reportRows As New Collection
    Dim headerRow(1 To 6) As Variant

    ' Headers
    headerRow(1) = "Rule Name"
    headerRow(2) = "Type"
    headerRow(3) = "Current Exposure"
    headerRow(4) = "Warning Level"
    headerRow(5) = "Breach Level"
    headerRow(6) = "Status"
    reportRows.Add headerRow

    Dim ruleKey As Variant
    Dim res As Variant

    If port.exposureLimits Is Nothing Then
        GenerateComplianceReport = CollectionTo2DArray(reportRows)
        Exit Function
    End If
    If port.exposureLimits.count = 0 Then
        GenerateComplianceReport = CollectionTo2DArray(reportRows)
        Exit Function
    End If

    ' Extract appropriate positions dictionary based on trades
    Dim dictToUse As Dictionary
    If Not port.trades Is Nothing Then
        If port.trades.count > 0 Then
            Set dictToUse = port.GetPositionsWithTrades()
        Else
            Set dictToUse = port.GetPositions(Date) ' Assuming Date if preTrade context isn't explicit here, but usually passed
        End If
    Else
        Set dictToUse = port.GetPositions(Date)
    End If

    For Each ruleKey In port.exposureLimits.Keys
        res = EvaluateSingleRule(port, dictToUse, CStr(ruleKey), port.exposureLimits(ruleKey))
        If Not IsEmpty(res) Then
            reportRows.Add CreateRow(res(0), res(1), res(2), res(3), res(4), res(5))
        End If
    Next ruleKey

    GenerateComplianceReport = CollectionTo2DArray(reportRows)

End Function

' Generates a consolidated matrix report for all portfolios
Public Function GenerateAggregatedComplianceReport(portfolios As Dictionary, preTrade As Boolean, Optional dt As Date = 0) As Variant

    If dt = 0 Then dt = Date ' Default to today if not provided

    Dim pKey As Variant
    Dim port As Portfolio
    Dim ruleKey As Variant

    ' 1. Identify all unique rules across all evaluated portfolios
    Dim uniqueRules As New Dictionary
    For Each pKey In portfolios
        Set port = portfolios(pKey)

        ' If preTrade is True, we only care about portfolios with trades
        Dim skipPort As Boolean
        skipPort = False
        If preTrade Then
            If port.trades Is Nothing Then
                skipPort = True
            ElseIf port.trades.count = 0 Then
                skipPort = True
            End If
        End If

        If Not skipPort Then
            If Not port.exposureLimits Is Nothing Then
                For Each ruleKey In port.exposureLimits.Keys
                    If Not uniqueRules.Exists(ruleKey) Then
                        uniqueRules.Add ruleKey, ruleKey
                    End If
                Next ruleKey
            End If
        End If
    Next pKey

    If uniqueRules.count = 0 Then
        GenerateAggregatedComplianceReport = Array()
        Exit Function
    End If

    ' Determine number of rows (only counting non-skipped portfolios)
    Dim nRows As Long: nRows = 1
    For Each pKey In portfolios
        Set port = portfolios(pKey)
        If preTrade Then
            If Not port.trades Is Nothing Then
                If port.trades.count > 0 Then nRows = nRows + 1
            End If
        Else
            nRows = nRows + 1
        End If
    Next pKey

    If nRows = 1 Then ' Only header
        GenerateAggregatedComplianceReport = Array()
        Exit Function
    End If

    ' 2. Initialize Report Array
    Dim nCols As Long: nCols = 1 + (uniqueRules.count * 3)
    Dim report() As Variant
    ReDim report(1 To nRows, 1 To nCols)

    ' Build Headers
    report(1, 1) = "Fund Name"
    Dim c As Long: c = 2
    Dim rKey As Variant
    Dim colMap As New Dictionary ' ruleKey -> Start Column Index

    For Each rKey In uniqueRules.Keys
        report(1, c) = rKey & " Exp"
        report(1, c + 1) = rKey & " Limit"
        report(1, c + 2) = rKey & " Status"
        colMap.Add rKey, c
        c = c + 3
    Next rKey

    ' 3. Populate Data
    Dim r As Long: r = 2
    Dim res As Variant
    Dim limitBreach As Double
    Dim dictToUse As Dictionary

    For Each pKey In portfolios
        Set port = portfolios(pKey)

        Dim skipPort2 As Boolean
        skipPort2 = False
        If preTrade Then
            If port.trades Is Nothing Then
                skipPort2 = True
            ElseIf port.trades.count = 0 Then
                skipPort2 = True
            End If
        End If

        If Not skipPort2 Then
            report(r, 1) = port.Name

            ' Determine which dictionary to use
            If preTrade And port.trades.count > 0 Then
                Set dictToUse = port.GetPositionsWithTrades()
            Else
                Set dictToUse = port.GetPositions(dt)
            End If

            For Each rKey In uniqueRules.Keys
                c = colMap(rKey)

                If Not port.exposureLimits Is Nothing Then
                    If port.exposureLimits.Exists(rKey) Then
                        limitBreach = port.exposureLimits(rKey)
                        res = EvaluateSingleRule(port, dictToUse, CStr(rKey), limitBreach)

                        If Not IsEmpty(res) Then
                            report(r, c) = res(2)     ' Exposure
                            report(r, c + 1) = res(4) ' Breach Limit
                            report(r, c + 2) = res(5) ' Status
                        Else
                            report(r, c) = "Err"
                            report(r, c + 1) = "Err"
                            report(r, c + 2) = "Err"
                        End If
                    Else
                        report(r, c) = "N/A"
                        report(r, c + 1) = "N/A"
                        report(r, c + 2) = "N/A"
                    End If
                Else
                    report(r, c) = "N/A"
                    report(r, c + 1) = "N/A"
                    report(r, c + 2) = "N/A"
                End If
            Next rKey
            r = r + 1
        End If
    Next pKey

    GenerateAggregatedComplianceReport = report
End Function


' Helper to evaluate a single rule.
' Returns a 0-based Array: (RuleName, Type, Exposure, WarningLimit, BreachLimit, Status)
' Returns Empty if ruleKey is not recognized.
Private Function EvaluateSingleRule(port As Portfolio, positionsDict As Dictionary, ruleKey As String, limitBreach As Double) As Variant

    Dim exp As Double
    Dim res As Variant

    Select Case UCase(Trim(ruleKey))
        Case "DELTAEQUITY_MIN"
            exp = port.GetEquityDelta(positionsDict)
            res = EvaluateThreshold(exp, limitBreach, "MIN")
            EvaluateSingleRule = Array("Delta Equity Minimum", "MIN", exp, res(0), limitBreach, res(1))

        Case "DELTAEQUITY_MAX"
            exp = port.GetEquityDelta(positionsDict)
            res = EvaluateThreshold(exp, limitBreach, "MAX")
            EvaluateSingleRule = Array("Delta Equity Maximum", "MAX", exp, res(0), limitBreach, res(1))

        Case "BONDS_MIN"
            exp = Calc_Bonds(positionsDict)
            res = EvaluateThreshold(exp, limitBreach, "MIN")
            EvaluateSingleRule = Array("Bonds Minimum", "MIN", exp, res(0), limitBreach, res(1))

        Case "BONDS_MAX"
            exp = Calc_Bonds(positionsDict)
            res = EvaluateThreshold(exp, limitBreach, "MAX")
            EvaluateSingleRule = Array("Bonds Maximum", "MAX", exp, res(0), limitBreach, res(1))

        Case "ALTERNATIVE_MAX"
            exp = Calc_Alternative(positionsDict)
            res = EvaluateThreshold(exp, limitBreach, "MAX")
            EvaluateSingleRule = Array("Alternative Maximum", "MAX", exp, res(0), limitBreach, res(1))

        Case "RISKOFF_MIN"
            exp = Calc_RiskOff(positionsDict)
            res = EvaluateThreshold(exp, limitBreach, "MIN")
            EvaluateSingleRule = Array("Risk Off Minimum", "MIN", exp, res(0), limitBreach, res(1))

        Case "STRUCTURED_MAX"
            exp = Calc_StructuredProducts(positionsDict)
            res = EvaluateThreshold(exp, limitBreach, "MAX")
            EvaluateSingleRule = Array("Structured Products Maximum", "MAX", exp, res(0), limitBreach, res(1))

        Case Else
            EvaluateSingleRule = Empty
    End Select

End Function


Private Function CreateRow(v1 As Variant, v2 As Variant, v3 As Variant, v4 As Variant, v5 As Variant, v6 As Variant) As Variant
    Dim arr(1 To 6) As Variant
    arr(1) = v1
    arr(2) = v2
    arr(3) = v3
    arr(4) = v4
    arr(5) = v5
    arr(6) = v6
    CreateRow = arr
End Function

' Converts a Collection of 1D arrays into a 2D Variant array
Private Function CollectionTo2DArray(col As Collection) As Variant
    If col.count = 0 Then
        CollectionTo2DArray = Array()
        Exit Function
    End If

    Dim r As Long, c As Long
    Dim numRows As Long, numCols As Long
    numRows = col.count
    numCols = UBound(col.Item(1)) ' Assumes all items are 1-based arrays of same length

    Dim result() As Variant
    ReDim result(1 To numRows, 1 To numCols)

    Dim rowArr As Variant
    For r = 1 To numRows
        rowArr = col.Item(r)
        For c = 1 To numCols
            result(r, c) = rowArr(c) ' Assuming rowArr is 1-based from Array() function or explicit dim
        Next c
    Next r

    CollectionTo2DArray = result
End Function

' Evaluates exposure against a breach limit and returns an Array(WarningLevel, Status)
Private Function EvaluateThreshold(exposure As Double, breachLimit As Double, ruleType As String) As Variant
    Dim warningLimit As Double
    Dim status As String

    If ruleType = "MAX" Then
        warningLimit = breachLimit * 0.9
        If exposure > breachLimit Then
            status = "BREACH"
        ElseIf exposure > warningLimit Then
            status = "WARNING"
        Else
            status = "PASS"
        End If
    ElseIf ruleType = "MIN" Then
        warningLimit = breachLimit * 1.1
        If exposure < breachLimit Then
            status = "BREACH"
        ElseIf exposure < warningLimit Then
            status = "WARNING"
        Else
            status = "PASS"
        End If
    Else
        status = "UNKNOWN"
        warningLimit = 0
    End If

    EvaluateThreshold = Array(warningLimit, status)
End Function

' ---------------------------------------------------------
' Rule Calculation Helpers
' ---------------------------------------------------------

Private Function Calc_Bonds(positionsDict As Dictionary) As Double
    Dim sum As Double
    Dim key As Variant
    Dim pos As Position
    Dim inst As Instrument

    If positionsDict Is Nothing Then Exit Function

    For Each key In positionsDict
        Set pos = positionsDict(key)
        Set inst = pos.posInstrument

        If Not inst Is Nothing Then
            If LCase(Trim(inst.assetClass)) = "bond" And LCase(Trim(inst.category)) <> "certificates" Then
                sum = sum + pos.weight
            End If
        End If
    Next key

    Calc_Bonds = sum
End Function

Private Function Calc_Alternative(positionsDict As Dictionary) As Double
    Dim sum As Double
    Dim key As Variant
    Dim pos As Position
    Dim inst As Instrument

    If positionsDict Is Nothing Then Exit Function

    For Each key In positionsDict
        Set pos = positionsDict(key)
        Set inst = pos.posInstrument

        If Not inst Is Nothing Then
            If LCase(Trim(inst.category)) = "alternatives investment" Then
                sum = sum + pos.weight
            End If
        End If
    Next key

    Calc_Alternative = sum
End Function

Private Function Calc_RiskOff(positionsDict As Dictionary) As Double
    Dim sum As Double
    Dim key As Variant
    Dim pos As Position
    Dim inst As Instrument
    Dim isBond As Boolean

    If positionsDict Is Nothing Then Exit Function

    For Each key In positionsDict
        Set pos = positionsDict(key)
        Set inst = pos.posInstrument

        If Not inst Is Nothing Then
            isBond = (LCase(Trim(inst.assetClass)) = "bond" And LCase(Trim(inst.category)) <> "certificates")

            If isBond Or LCase(Trim(inst.assetClass)) = "real estate" Or LCase(Trim(inst.assetClass)) = "cash" Then
                sum = sum + pos.weight
            End If
        End If
    Next key

    Calc_RiskOff = sum
End Function

Private Function Calc_StructuredProducts(positionsDict As Dictionary) As Double
    Dim sum As Double
    Dim key As Variant
    Dim pos As Position
    Dim inst As Instrument

    If positionsDict Is Nothing Then Exit Function

    For Each key In positionsDict
        Set pos = positionsDict(key)
        Set inst = pos.posInstrument

        If Not inst Is Nothing Then
            If LCase(Trim(inst.category)) = "certificates" Then
                sum = sum + pos.weight
            End If
        End If
    Next key

    Calc_StructuredProducts = sum
End Function
