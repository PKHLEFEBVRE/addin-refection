Option Explicit

Public Function GetExposure(port As Portfolio, key1 As String, Optional key2 As String = "") As Dictionary

    Dim exposure As New Dictionary
    Dim pos As Position
    Dim inst As Instrument
    Dim posKey As Variant
    Dim val1 As String, val2 As String
    Dim compositeKey As String
    Dim weight As Double

    If port.positions Is Nothing Then
        Set GetExposure = exposure
        Exit Function
    End If

    For Each posKey In port.positions
        Set pos = port.positions(posKey)
        Set inst = pos.posInstrument

        ' Get value for key1
        val1 = GetValueForExposure(pos, inst, key1)

        ' Get value for key2 if provided
        If key2 <> "" Then
            val2 = GetValueForExposure(pos, inst, key2)
            compositeKey = val1 & "|" & val2
        Else
            compositeKey = val1
        End If

        ' Sum weights
        ' If weight is not calculated, fallback to amountBaseCur / AUM if AUM > 0
        weight = pos.weight

        If exposure.Exists(compositeKey) Then
            exposure(compositeKey) = exposure(compositeKey) + weight
        Else
            exposure.Add compositeKey, weight
        End If

    Next posKey

    Set GetExposure = exposure

End Function

Private Function GetValueForExposure(pos As Position, inst As Instrument, key As String) As String

    Dim val As Variant

    ' Try to get from Instrument first
    If Not inst Is Nothing Then
        If HasMember(inst, key) Then
            val = CallByName(inst, key, VbGet)
            GetValueForExposure = CStr(val)
            Exit Function
        End If
    End If

    ' Then try Position
    If HasMember(pos, key) Then
        val = CallByName(pos, key, VbGet)
        GetValueForExposure = CStr(val)
        Exit Function
    End If

    ' Fallback
    GetValueForExposure = "N/A"

End Function

Private Function HasMember(obj As Object, memberName As String) As Boolean
    On Error Resume Next
    Call CallByName(obj, memberName, VbGet)
    HasMember = (Err.Number = 0)
    Err.Clear
End Function


Public Function GetAggregatedExposure(portfolios As Dictionary, key1 As String, Optional key2 As String = "") As Variant

    Dim exposureDicts As New Dictionary
    Dim uniqueKeys As New Dictionary
    Dim portKey As Variant
    Dim port As Portfolio
    Dim expDict As Dictionary
    Dim expKey As Variant
    Dim result(0 To 1) As Variant ' 0: dict of dicts, 1: unique keys dict

    For Each portKey In portfolios
        Set port = portfolios(portKey)

        Set expDict = GetExposure(port, key1, key2)
        exposureDicts.Add portKey, expDict

        For Each expKey In expDict
            If Not uniqueKeys.Exists(expKey) Then
                uniqueKeys.Add expKey, expKey
            End If
        Next expKey
    Next portKey

    Set result(0) = exposureDicts
    Set result(1) = uniqueKeys

    GetAggregatedExposure = result

End Function

Public Function GenerateDiffPortfolios(portfolios As Dictionary, benchmarks As Dictionary) As Dictionary

    Dim diffPortfolios As New Dictionary
    Dim key As Variant
    Dim port As Portfolio, bench As Portfolio, diffPort As Portfolio

    For Each key In portfolios
        Set port = portfolios(key)
        If benchmarks.Exists(key) Then
            Set bench = benchmarks(key)
            Set diffPort = PortfolioFactory.CreateDiffPortfolio(port, bench)
            diffPortfolios.Add diffPort.Name, diffPort
        End If
    Next key

    Set GenerateDiffPortfolios = diffPortfolios

End Function

Public Function GetExposureReport(portfolios As Dictionary, key1 As String, Optional key2 As String = "") As Variant

    Dim aggData As Variant
    aggData = GetAggregatedExposure(portfolios, key1, key2)

    Dim expDicts As Dictionary
    Dim keys As Dictionary

    Set expDicts = aggData(0)
    Set keys = aggData(1)

    Dim table As Variant
    table = CommunFactory.TurnExposureDictsToTable(expDicts, keys)

    GetExposureReport = table

End Function
