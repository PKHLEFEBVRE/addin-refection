Public Function CreateInventoryCsvPath(ByVal dt As Date, ByVal mandateId As String) As String
    
    Dim fileName As String, fullPath As String
    
    dt = Application.WorksheetFunction.WorkDay(dt, -1)
    
    fileName = "Inventaire_" & mandateId & "_" & CStr(Format(dt, "YYYYMMDD")) & ".csv"
    fullPath = rootDirInfinInventories & "\" & CStr(Year(dt)) & "\" & makeItTwoDigits(Month(dt)) & "\" & makeItTwoDigits(Day(dt)) & "\" & fileName
    
    CreateInventoryCsvPath = fullPath
    
End Function


Public Function makeItTwoDigits(ByVal inte As Integer) As String

    If inte < 10 Then
        makeItTwoDigits = "0" & CStr(inte)
    Else
        makeItTwoDigits = CStr(inte)
    End If

End Function

    


Public Function CreateInstrumentsReferentialPath(ByVal dt As Date) As String
    
    Dim fileName As String, fullPath As String

    fileName = "INFIN_referential_securities_" & CStr(Format(dt, "YYYYMMDD")) & ".xlsx"
    fullPath = rootDirReferential & "\" & CStr(Year(dt)) & "\" & makeItTwoDigits(Month(dt)) & "\" & fileName
    
    CreateInstrumentsReferentialPath = fullPath
    
End Function
