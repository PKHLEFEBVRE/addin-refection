Option Explicit

Public Static Function GetPortfoliosFromFile(ByVal path As String, ByVal dt As Date) As Dictionary

    Dim lines() As Variant
    lines = ExcelFactory.GetLinesFromExcelFile(path, , True)
    If CommunFactory.isArrayEmpty(lines) Then
        MsgBox "No portfolio referential found at:" & vbCr & vbCr & path, vbCritical, "Fatal"
        Set GetPortfoliosFromFile = New Dictionary
        Exit Function
    End If

    Dim port As Portfolio
    Dim portfolios As Dictionary
    Set portfolios = New Dictionary

    Dim i As Integer, line As Variant, key As String, inventoryId As String

    Dim columnsDict As Dictionary
    Set columnsDict = createColumnsDictFromInventory(lines(1))

    For i = LBound(lines) + 1 To UBound(lines)  'remove headers

        line = lines(i)

        ' Determine Key: Use InfinId if available, else Name
        key = ""
        If columnsDict.Exists("InfinId") Then
            key = CStr(line(columnsDict("InfinId")))
        End If

        If key = "" And columnsDict.Exists("Name") Then
            key = CStr(line(columnsDict("Name")))
        End If

        If key <> "" Then
            Set port = CreatePortfolio(line, columnsDict)

            ' Inventory is NOT loaded here anymore. It must be loaded on demand via LoadPositionsForType.

            If Not portfolios.Exists(key) Then
                portfolios.Add key, port
            Else
                Debug.Print "Duplicate portfolio key: " & key
            End If
        End If

    Next i

    Set GetPortfoliosFromFile = portfolios

End Function

Private Function createColumnsDictFromInventory(ByVal line As Variant) As Dictionary

    Dim c As Integer, columnsNumber As Dictionary
    Set columnsNumber = New Dictionary

    For c = 1 To UBound(line)
        If Not columnsNumber.Exists(line(c)) Then
            columnsNumber.Add line(c), c
        End If
    Next c

    Set createColumnsDictFromInventory = columnsNumber

End Function

Public Static Function CreatePortfolio(ByVal line As Variant, ByVal columnsDict As Dictionary) As Portfolio

    Dim port As Portfolio
    Set port = New Portfolio

    ' String properties
    If columnsDict.Exists("Name") Then port.Name = CStr(Replace(line(columnsDict("Name")), """", ""))
    If columnsDict.Exists("comment") Then port.comment = CStr(Replace(line(columnsDict("comment")), """", ""))
    If columnsDict.Exists("InfinId") Then port.InfinId = CStr(Replace(line(columnsDict("InfinId")), """", ""))
    If columnsDict.Exists("insuranceContract") Then port.insuranceContract = CStr(Replace(line(columnsDict("insuranceContract")), """", ""))
    If columnsDict.Exists("accountName") Then port.accountName = CStr(Replace(line(columnsDict("accountName")), """", ""))
    If columnsDict.Exists("client") Then port.client = CStr(Replace(line(columnsDict("client")), """", ""))
    If columnsDict.Exists("insurer") Then port.insurer = CStr(Replace(line(columnsDict("insurer")), """", ""))
    If columnsDict.Exists("custodian") Then port.custodian = CStr(Replace(line(columnsDict("custodian")), """", ""))
    If columnsDict.Exists("sales") Then port.sales = CStr(Replace(line(columnsDict("sales")), """", ""))

    ' Mapped properties change: AccountType -> MandatType, Type -> FundType
    If columnsDict.Exists("Type") Then port.FundType = CStr(Replace(line(columnsDict("Type")), """", ""))
    If columnsDict.Exists("AccountType") Then port.MandatType = CStr(Replace(line(columnsDict("AccountType")), """", ""))

    If columnsDict.Exists("AVType") Then port.AVType = CStr(Replace(line(columnsDict("AVType")), """", ""))
    If columnsDict.Exists("ccy") Then port.ccy = CStr(Replace(line(columnsDict("ccy")), """", ""))
    If columnsDict.Exists("profile") Then port.profile = CStr(Replace(line(columnsDict("profile")), """", ""))
    If columnsDict.Exists("composition") Then port.composition = CStr(Replace(line(columnsDict("composition")), """", ""))
    If columnsDict.Exists("targetProfile") Then port.targetProfile = CStr(Replace(line(columnsDict("targetProfile")), """", ""))
    If columnsDict.Exists("targetComposition") Then port.targetComposition = CStr(Replace(line(columnsDict("targetComposition")), """", ""))
    If columnsDict.Exists("lastComputed") Then port.lastComputed = CStr(Replace(line(columnsDict("lastComputed")), """", ""))

    ' Date properties
    If columnsDict.Exists("creationDate") Then
        If CStr(line(columnsDict("creationDate"))) <> "" Then
            port.creationDate = CDate(CStr(Replace(line(columnsDict("creationDate")), """", "")))
        End If
    End If

    ' Double properties
    If columnsDict.Exists("CashAndTreso") Then
        If CStr(line(columnsDict("CashAndTreso"))) <> "" Then
            port.CashAndTreso = CDbl(Replace(line(columnsDict("CashAndTreso")), ".", Application.International(xlDecimalSeparator)))
        End If
    End If

    If columnsDict.Exists("AUM") Then
        If CStr(line(columnsDict("AUM"))) <> "" Then
            port.AUM = CDbl(Replace(line(columnsDict("AUM")), ".", Application.International(xlDecimalSeparator)))
        End If
    End If

    Set CreatePortfolio = port

End Function


Public Sub LoadPositionsForType(portfolios As Dictionary, typeFilter As String, dt As Date)

    Dim key As Variant
    Dim port As Portfolio

    For Each key In portfolios
        Set port = portfolios(key)
        If LCase(port.FundType) = LCase(typeFilter) Then
            port.GetPositions dt
        End If
    Next key

End Sub


Public Static Function CreateDiffPortfolio(port As Portfolio, bench As Portfolio) As Portfolio

    Dim diffPort As Portfolio
    Set diffPort = New Portfolio

    diffPort.Name = port.Name & " vs " & bench.Name
    diffPort.FundType = "Difference"
    diffPort.MandatType = port.MandatType
    diffPort.ccy = port.ccy

    ' Initialize positions dict
    Set diffPort.positions = New Dictionary

    Dim pPos As Position, bPos As Position, newPos As Position
    Dim key As Variant

    ' Add all positions from Portfolio (Positive Weight)
    If Not port.positions Is Nothing Then
        For Each key In port.positions
            Set pPos = port.positions(key)
            Set newPos = pPos.Clone()
            ' Ensure clone keeps weight (Clone method does)
            diffPort.positions.Add newPos.posID, newPos
        Next key
    End If

    ' Subtract positions from Benchmark
    If Not bench.positions Is Nothing Then
        For Each key In bench.positions
            Set bPos = bench.positions(key)

            If diffPort.positions.Exists(bPos.posID) Then
                ' Position exists, update weight
                diffPort.positions(bPos.posID).weight = diffPort.positions(bPos.posID).weight - bPos.weight
            Else
                ' Position does not exist, add with negative weight
                Set newPos = bPos.Clone()
                newPos.weight = -bPos.weight
                diffPort.positions.Add newPos.posID, newPos
            End If
        Next key
    End If

    Set CreateDiffPortfolio = diffPort

End Function

Public Sub EnrichPortfoliosFromDashboard(portfolios As Dictionary, ws As Worksheet)

    Dim data As Variant
    data = ws.UsedRange.Value

    If Not IsArray(data) Then Exit Sub

    Dim r As Long, c As Long
    Dim headers As Dictionary
    Set headers = New Dictionary

    ' Parse Headers (Row 1)
    For c = 1 To UBound(data, 2)
        If Not IsError(data(1, c)) Then
            headers(CStr(data(1, c))) = c
        End If
    Next c

    Dim key As String
    Dim port As Portfolio
    Dim val As Variant

    ' Loop Data
    For r = 2 To UBound(data, 1)

        ' Identify Portfolio (Key)
        ' Assuming "Account name" is the primary key used in the Dashboard sheet
        key = ""
        If headers.Exists("Account name") Then
            key = CStr(data(r, headers("Account name")))
        End If

        ' Find Portfolio
        ' Note: portfolios dict might be keyed by InfinId or Name. We check both if possible or rely on Name.
        ' Since GetPortfoliosFromFile uses Name if InfinId is missing, we try to match Name here.
        ' However, the main dict keys might be IDs. We might need to scan if direct lookup fails.

        Set port = Nothing
        ' Try direct lookup if key matches dictionary key logic
        If portfolios.Exists(key) Then
            Set port = portfolios(key)
        Else
            ' Fallback: Scan by Name or AccountName
            Dim pKey As Variant
            For Each pKey In portfolios
                If LCase(portfolios(pKey).accountName) = LCase(key) Or LCase(portfolios(pKey).Name) = LCase(key) Then
                    Set port = portfolios(pKey)
                    Exit For
                End If
            Next pKey
        End If

        If Not port Is Nothing Then
            ' Update Attributes
            If headers.Exists("Client") Then port.client = CStr(data(r, headers("Client")))

            If headers.Exists("Creation date") Then
                If IsDate(data(r, headers("Creation date"))) Then
                    port.creationDate = CDate(data(r, headers("Creation date")))
                End If
            End If

            If headers.Exists("Insurer") Then port.insurer = CStr(data(r, headers("Insurer")))
            If headers.Exists("Custodian") Then port.custodian = CStr(data(r, headers("Custodian")))
            If headers.Exists("Sales") Then port.sales = CStr(data(r, headers("Sales")))

            If headers.Exists("Account type") Then port.MandatType = CStr(data(r, headers("Account type")))
            If headers.Exists("AV Type") Then port.AVType = CStr(data(r, headers("AV Type")))
            If headers.Exists("CCY") Then port.ccy = CStr(data(r, headers("CCY")))
            If headers.Exists("Profile") Then port.profile = CStr(data(r, headers("Profile")))
            If headers.Exists("Composition") Then port.composition = CStr(data(r, headers("Composition")))
            If headers.Exists("Target Profile") Then port.targetProfile = CStr(data(r, headers("Target Profile")))
            If headers.Exists("Target Composition") Then port.targetComposition = CStr(data(r, headers("Target Composition")))
        End If

    Next r

End Sub
