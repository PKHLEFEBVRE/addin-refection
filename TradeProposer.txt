Option Explicit

Public Function ProposeTradesAsPortfolio(target As Portfolio, bench As Portfolio) As Portfolio
    ' Generate actionable trade proposals to align a target portfolio with its benchmark.
    ' Focuses on the single largest discrepancy (> 3%) across the intersection of Asset Class and Category.
    ' Returns a Portfolio containing a synthetic Position representing the required trade.

    Dim tradesPort As New Portfolio
    tradesPort.Name = "Proposed Trades for " & target.Name

    Dim threshold As Double
    threshold = 0.03 ' 3% threshold for "large" discrepancies

    ' 1. Create Difference Portfolio (Target - Benchmark)
    Dim diffPort As Portfolio
    Set diffPort = PortfolioFactory.CreateDiffPortfolio(target, bench)

    ' 2. Calculate exposures of the difference portfolio (Intersection of Asset Class and Category)
    ' Positive exposure means Target > Benchmark (Overweight -> Need to Sell)
    ' Negative exposure means Target < Benchmark (Underweight -> Need to Buy)

    Dim intersectionDiffs As Dictionary
    Set intersectionDiffs = PositionsReportsFactory.GetExposure(diffPort, "assetClass", "category")

    ' 3. Find the single largest discrepancy across the intersection keys
    Dim maxDiff As Double
    Dim maxKey As String
    Dim key As Variant
    Dim diff As Double

    maxDiff = 0
    maxKey = ""

    For Each key In intersectionDiffs.Keys
        diff = intersectionDiffs(key)
        If Abs(diff) > Abs(maxDiff) Then
            maxDiff = diff
            maxKey = CStr(key)
        End If
    Next key

    ' 4. Generate the single proposal if it exceeds the threshold
    ' Note: diff is Target - Benchmark.
    ' To align target with benchmark, the trade weight is -maxDiff.
    Dim tradeWeight As Double
    tradeWeight = -maxDiff

    If Abs(tradeWeight) >= threshold Then
        Dim pos As New Position
        Dim targetInst As Instrument
        Dim targetId As String
        Dim targetName As String

        ' Find the most prominent security to trade in this intersection
        If tradeWeight > 0 Then
            ' Need to Buy: Find the biggest security in the benchmark matching the key
            FindProminentSecurity bench, maxKey, targetId, targetName, targetInst
        Else
            ' Need to Sell: Find the biggest security in the target portfolio matching the key
            FindProminentSecurity target, maxKey, targetId, targetName, targetInst
        End If

        ' If a security couldn't be found (unlikely but possible), use a placeholder
        If targetId = "" Then
            targetId = "N/A"
            targetName = "Generic " & maxKey
        End If

        pos.posID = targetId
        pos.name = targetName
        pos.weight = tradeWeight ' Positive weight means Buy, Negative means Sell
        pos.account = maxKey ' Store the intersection key for email reporting
        Set pos.posInstrument = targetInst

        tradesPort.positions.Add pos.posID, pos
    End If

    Set ProposeTradesAsPortfolio = tradesPort
End Function

Private Sub FindProminentSecurity(port As Portfolio, intersectionKey As String, ByRef outId As String, ByRef outName As String, ByRef outInst As Instrument)
    Dim posKey As Variant
    Dim pos As Position
    Dim inst As Instrument
    Dim val1 As String, val2 As String
    Dim compositeKey As String
    Dim maxWeight As Double

    maxWeight = 0
    outId = ""
    outName = ""
    Set outInst = Nothing

    If port.positions Is Nothing Then Exit Sub

    For Each posKey In port.positions.Keys
        Set pos = port.positions(posKey)
        Set inst = pos.posInstrument

        val1 = "N/A"
        val2 = "N/A"

        If Not inst Is Nothing Then
            On Error Resume Next
            val1 = CStr(CallByName(inst, "assetClass", VbGet))
            If Err.Number <> 0 Then val1 = "N/A"
            Err.Clear

            val2 = CStr(CallByName(inst, "category", VbGet))
            If Err.Number <> 0 Then val2 = "N/A"
            Err.Clear
            On Error GoTo 0
        End If

        compositeKey = val1 & "|" & val2

        If compositeKey = intersectionKey Then
            If Abs(pos.weight) > maxWeight Then
                maxWeight = Abs(pos.weight)
                outId = pos.posID
                outName = pos.name
                Set outInst = inst
            End If
        End If
    Next posKey
End Sub

Public Sub GenerateAndEmailProposals(portfolios As Dictionary, Optional emailTo As String = "")
    ' Iterates through all portfolios, gets their benchmarks, finds the largest trade proposals,
    ' formats them into an HTML table, and sends an email via Outlook.

    Dim htmlBody As String
    Dim portKey As Variant
    Dim targetPortfolio As Portfolio
    Dim bench As Portfolio
    Dim tradesPort As Portfolio
    Dim posKey As Variant
    Dim tradePos As Position
    Dim hasProposals As Boolean
    Dim action As String
    Dim actionColor As String

    hasProposals = False

    ' Construct HTML Header
    htmlBody = "<html><body style='font-family: Arial, sans-serif;'>"
    htmlBody = htmlBody & "<h2>Benchmark Trade Proposals</h2>"
    htmlBody = htmlBody & "<p>The following table lists the single largest recommended trade (>3% discrepancy) to align each portfolio with its composition benchmark.</p>"
    htmlBody = htmlBody & "<table border='1' cellpadding='5' cellspacing='0' style='border-collapse: collapse;'>"
    htmlBody = htmlBody & "<tr style='background-color: #f2f2f2;'>"
    htmlBody = htmlBody & "<th>Portfolio Name</th>"
    htmlBody = htmlBody & "<th>Action</th>"
    htmlBody = htmlBody & "<th>Weight to Adjust</th>"
    htmlBody = htmlBody & "<th>Asset Class | Category</th>"
    htmlBody = htmlBody & "<th>Security ID</th>"
    htmlBody = htmlBody & "<th>Security Name</th>"
    htmlBody = htmlBody & "</tr>"

    ' Process Portfolios
    For Each portKey In portfolios.Keys
        Set targetPortfolio = portfolios(portKey)

        ' Retrieve benchmark from portfolio dictionary (do not compute)
        On Error Resume Next
        Set bench = Nothing
        If Not targetPortfolio.benchmark Is Nothing Then
            If targetPortfolio.benchmark.Exists("composition") Then
                Set bench = targetPortfolio.benchmark("composition")
            End If
        End If
        On Error GoTo 0

        If Not bench Is Nothing Then
            Set tradesPort = ProposeTradesAsPortfolio(targetPortfolio, bench)

            ' Save trades to benchmark dictionary
            If targetPortfolio.benchmark.Exists("trades") Then
                Set targetPortfolio.benchmark("trades") = tradesPort
            Else
                targetPortfolio.benchmark.Add "trades", tradesPort
            End If

            ' If there's a trade proposal
            If Not tradesPort Is Nothing Then
                If tradesPort.positions.Count > 0 Then
                    hasProposals = True

                    ' Only one position expected
                    For Each posKey In tradesPort.positions.Keys
                        Set tradePos = tradesPort.positions(posKey)

                        ' Determine Action and Color
                        If tradePos.weight > 0 Then
                            action = "Buy"
                            actionColor = "green"
                        Else
                            action = "Sell"
                            actionColor = "red"
                        End If

                        ' Add row
                        htmlBody = htmlBody & "<tr>"
                        htmlBody = htmlBody & "<td>" & targetPortfolio.Name & "</td>"
                        htmlBody = htmlBody & "<td style='color:" & actionColor & "; font-weight:bold;'>" & action & "</td>"
                        htmlBody = htmlBody & "<td>" & Format(Abs(tradePos.weight), "0.00%") & "</td>"
                        htmlBody = htmlBody & "<td>" & tradePos.account & "</td>" ' intersection key stored in account
                        htmlBody = htmlBody & "<td>" & tradePos.posID & "</td>"
                        htmlBody = htmlBody & "<td>" & tradePos.name & "</td>"
                        htmlBody = htmlBody & "</tr>"
                    Next posKey
                End If
            End If
        Else
            Debug.Print "Warning: 'composition' benchmark not found for portfolio " & targetPortfolio.Name
        End If
    Next portKey

    ' Finish HTML
    htmlBody = htmlBody & "</table>"
    If Not hasProposals Then
        htmlBody = htmlBody & "<p><i>No large discrepancies (>3%) found for any portfolio.</i></p>"
    End If
    htmlBody = htmlBody & "</body></html>"

    ' Send Email via Outlook COM
    On Error Resume Next
    Dim outApp As Object
    Dim outMail As Object

    Set outApp = CreateObject("Outlook.Application")
    If outApp Is Nothing Then
        MsgBox "Failed to create Outlook application object. Is Outlook installed?", vbCritical, "Email Error"
        Exit Sub
    End If

    Set outMail = outApp.CreateItem(0) ' 0 = olMailItem

    With outMail
        If emailTo <> "" Then .To = emailTo
        .Subject = "Benchmark Trade Proposals Report - " & Format(Date, "YYYY-MM-DD")
        .HTMLBody = htmlBody
        .Display ' Display rather than .Send automatically for safety/review
    End With

    Set outMail = Nothing
    Set outApp = Nothing
    On Error GoTo 0

End Sub
