Option Explicit

Public Function ProposeTradesAsPortfolio(target As Portfolio, bench As Portfolio) As Portfolio
    ' Generate actionable trade proposals to align a target portfolio with its benchmark.
    ' Focuses on large discrepancies (> 3%) across Asset Class, Geography, and Category.
    ' Returns a Portfolio containing synthetic Positions representing the required trades.

    Dim tradesPort As New Portfolio
    tradesPort.Name = "Proposed Trades for " & target.Name

    Dim threshold As Double
    threshold = 0.03 ' 3% threshold for "large" discrepancies

    ' 1. Check Asset Class discrepancies
    Dim assetClassDiffs As Dictionary
    Set assetClassDiffs = GetExposureDifferences(target, bench, "assetClass")
    GenerateProposals assetClassDiffs, "AssetClass", threshold, tradesPort

    ' 2. Check Geography discrepancies
    Dim geoDiffs As Dictionary
    Set geoDiffs = GetExposureDifferences(target, bench, "geography")
    GenerateProposals geoDiffs, "Geography", threshold, tradesPort

    ' 3. Check Category discrepancies
    Dim catDiffs As Dictionary
    Set catDiffs = GetExposureDifferences(target, bench, "category")
    GenerateProposals catDiffs, "Category", threshold, tradesPort

    Set ProposeTradesAsPortfolio = tradesPort
End Function

Private Function GetExposureDifferences(target As Portfolio, bench As Portfolio, propertyName As String) As Dictionary
    Dim targetExp As Dictionary
    Dim benchExp As Dictionary
    Dim diffs As New Dictionary
    Dim key As Variant
    Dim targetWeight As Double
    Dim benchWeight As Double
    Dim diff As Double

    ' Calculate exposures manually to avoid external dependencies
    Set targetExp = CalculateExposure(target, propertyName)
    Set benchExp = CalculateExposure(bench, propertyName)

    ' Get all unique keys from both portfolios
    Dim allKeys As New Dictionary
    For Each key In targetExp.Keys: allKeys(key) = 1: Next key
    For Each key In benchExp.Keys: allKeys(key) = 1: Next key

    ' Calculate diffs (Benchmark - Target) => Positive means we need to Buy, Negative means Sell
    For Each key In allKeys.Keys
        targetWeight = 0
        If targetExp.Exists(key) Then targetWeight = targetExp(key)

        benchWeight = 0
        If benchExp.Exists(key) Then benchWeight = benchExp(key)

        diff = benchWeight - targetWeight
        If Abs(diff) > 0.0001 Then ' Avoid floating point noise
            diffs.Add CStr(key), diff
        End If
    Next key

    Set GetExposureDifferences = diffs
End Function

Private Function CalculateExposure(port As Portfolio, propertyName As String) As Dictionary
    Dim exp As New Dictionary
    Dim posKey As Variant
    Dim pos As Position
    Dim inst As Instrument
    Dim propValue As String
    Dim weight As Double

    If port.positions Is Nothing Then
        Set CalculateExposure = exp
        Exit Function
    End If

    For Each posKey In port.positions.Keys
        Set pos = port.positions(posKey)
        Set inst = pos.posInstrument
        propValue = "N/A"
        weight = pos.weight

        ' Retrieve property value dynamically from Instrument
        If Not inst Is Nothing Then
            On Error Resume Next
            propValue = CStr(CallByName(inst, propertyName, VbGet))
            If Err.Number <> 0 Then propValue = "N/A"
            On Error GoTo 0
        End If

        ' Only aggregate valid dimensions
        If propValue <> "N/A" And propValue <> "" Then
            If exp.Exists(propValue) Then
                exp(propValue) = exp(propValue) + weight
            Else
                exp.Add propValue, weight
            End If
        End If
    Next posKey

    Set CalculateExposure = exp
End Function

Private Sub GenerateProposals(diffs As Dictionary, dimensionName As String, threshold As Double, ByRef tradesPort As Portfolio)
    Dim key As Variant
    Dim diff As Double
    Dim pos As Position
    Dim syntheticId As String

    For Each key In diffs.Keys
        diff = diffs(key)

        ' Only propose if discrepancy exceeds the threshold
        If Abs(diff) >= threshold Then
            ' Create a synthetic position representing the trade
            Set pos = New Position
            syntheticId = dimensionName & "::" & CStr(key)

            pos.posID = syntheticId
            pos.name = "Adjust " & dimensionName & " exposure to " & CStr(key)
            pos.weight = diff ' Positive weight means Buy, Negative means Sell

            tradesPort.positions.Add pos.posID, pos
        End If
    Next key
End Sub
