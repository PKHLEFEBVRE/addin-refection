Option Explicit

' Proposes a trade to match the benchmark weight of a specifically selected ISIN
' Automatically handles cash buffer and Money Market funding if required.
Public Function ProposeSingleLineTrade(port As Portfolio, bench As Portfolio, selectedISIN As String) As Dictionary

    Dim proposedTrades As New Dictionary

    Dim bPos As Position
    Dim pPos As Position
    Dim targetWeight As Double: targetWeight = 0
    Dim currentWeight As Double: currentWeight = 0
    Dim tradeWeight As Double
    Dim tradeAmount As Double

    Dim config As Configuration
    Set config = Data.GetConfiguration()

    ' 1. Find Target Weight in Benchmark
    If Not bench.positions Is Nothing Then
        ' Note: benchmarking logic sometimes uses InfinID, sometimes ISIN as the Dictionary key.
        ' We iterate to be safe and match by the actual ISIN property.
        Dim bKey As Variant
        For Each bKey In bench.positions
            If bench.positions(bKey).isin = selectedISIN Then
                Set bPos = bench.positions(bKey)
                targetWeight = bPos.weight
                Exit For
            End If
        Next bKey
    End If

    ' (Removed early exit if bPos is Nothing to allow selling non-benchmark assets to 0%)

    ' 2. Find Current Weight in Portfolio
    If Not port.positions Is Nothing Then
        Dim pKey As Variant
        For Each pKey In port.positions
            If port.positions(pKey).isin = selectedISIN Then
                Set pPos = port.positions(pKey)
                currentWeight = pPos.weight
                Exit For
            End If
        Next pKey
    End If

    ' 3. Calculate Trade Amount
    tradeWeight = targetWeight - currentWeight

    If Abs(tradeWeight) < 0.0001 Then
        ' Negligible difference, no trade needed
        Set ProposeSingleLineTrade = proposedTrades
        Exit Function
    End If

    tradeAmount = tradeWeight * port.AUM

    ' 4. Create Asset Trade Position
    Dim assetTrade As Position
    If Not pPos Is Nothing Then
        ' Base trade on existing position to inherit CCY, FX, etc.
        Set assetTrade = pPos.Clone()
    ElseIf Not bPos Is Nothing Then
        ' Base trade on benchmark position if we don't own it yet
        Set assetTrade = bPos.Clone()
        assetTrade.account = port.Name ' Align with our portfolio
    Else
        ' This should never be reached if called correctly from ProposeLargestDiscrepancyTrade,
        ' as the ISIN must exist in one or the other.
        MsgBox "Cannot propose trade: ISIN " & selectedISIN & " found in neither portfolio nor benchmark.", vbCritical
        Set ProposeSingleLineTrade = proposedTrades
        Exit Function
    End If

    ' Calculate exact Quantity needed
    Dim priceFactor As Double
    If Not assetTrade.posInstrument Is Nothing Then
        priceFactor = assetTrade.posInstrument.lastClose * assetTrade.fx
        If assetTrade.posInstrument.quoteInPct Then priceFactor = priceFactor / 100
    Else
        ' Fallback to position data if instrument referential is missing
        priceFactor = assetTrade.price * assetTrade.fx * assetTrade.quoteInPct
    End If

    If priceFactor = 0 Then
        MsgBox "Cannot calculate quantity: price or fx is zero for " & selectedISIN & " in portfolio " & port.Name, vbCritical
        Set ProposeSingleLineTrade = proposedTrades
        Exit Function
    End If

    assetTrade.qty = tradeAmount / priceFactor
    assetTrade.amountBaseCur = tradeAmount
    assetTrade.weight = tradeWeight

    proposedTrades.Add assetTrade.posID, assetTrade

    ' 5. Handle Funding (Only if Buying)
    If tradeAmount > 0 Then

        Dim cashPos As Position
        Dim baseCashKey As Variant

        ' Find Base Cash (where pos.ccy = port.ccy, usually posID is the CCY code or "10")
        Dim isCashLine As Boolean
        For Each pKey In port.positions
            If UCase(Trim(port.positions(pKey).ccy)) = UCase(Trim(port.ccy)) Then
                isCashLine = False
                If port.positions(pKey).posID = "10" Then
                    isCashLine = True
                ElseIf Not port.positions(pKey).posInstrument Is Nothing Then
                    If LCase(Trim(port.positions(pKey).posInstrument.category)) = "cash" Then
                        isCashLine = True
                    End If
                End If

                If isCashLine Then
                    Set cashPos = port.positions(pKey)
                    Exit For
                End If
            End If
        Next pKey

        Dim currentCash As Double: currentCash = 0
        If Not cashPos Is Nothing Then currentCash = cashPos.amountBaseCur

        ' Require a 0.4% buffer to avoid overdraft
        Dim requiredBuffer As Double
        requiredBuffer = port.AUM * 0.004

        Dim availableCash As Double
        availableCash = currentCash - requiredBuffer

        If availableCash < tradeAmount Then
            ' Shortfall! Need to sell Money Market fund
            Dim shortfall As Double
            shortfall = tradeAmount
            If availableCash > 0 Then shortfall = tradeAmount - availableCash

            Dim mmPos As Position
            Dim mmISIN As String
            mmISIN = config.globalMoneyMarketISIN

            For Each pKey In port.positions
                If port.positions(pKey).isin = mmISIN Then
                    Set mmPos = port.positions(pKey)
                    Exit For
                End If
            Next pKey

            If mmPos Is Nothing Then
                MsgBox "Insufficient cash and no Money Market fund (" & mmISIN & ") found to fund this trade for portfolio " & port.Name, vbCritical, "Funding Error"
                ' Still return the asset trade, but user is warned
            Else
                If mmPos.amountBaseCur < shortfall Then
                    MsgBox "Warning: Selling ALL Money Market fund for portfolio " & port.Name & ", but still short by " & Format(shortfall - mmPos.amountBaseCur, "#,##0") & " " & port.ccy, vbExclamation, "Funding Warning"
                    shortfall = mmPos.amountBaseCur ' Sell maximum available
                End If

                Dim mmTrade As Position
                Set mmTrade = mmPos.Clone()

                Dim mmPriceFactor As Double
                If Not mmTrade.posInstrument Is Nothing Then
                    mmPriceFactor = mmTrade.posInstrument.lastClose * mmTrade.fx
                    If mmTrade.posInstrument.quoteInPct Then mmPriceFactor = mmPriceFactor / 100
                Else
                    mmPriceFactor = mmTrade.price * mmTrade.fx * mmTrade.quoteInPct
                End If

                If mmPriceFactor > 0 Then
                    mmTrade.qty = -shortfall / mmPriceFactor
                    mmTrade.amountBaseCur = -shortfall
                    mmTrade.weight = -shortfall / port.AUM
                    proposedTrades.Add mmTrade.posID & "_FUNDING", mmTrade
                End If
            End If
        End If

    End If

    Set ProposeSingleLineTrade = proposedTrades

End Function

' Evaluates all portfolios against their respective benchmarks for a selected ISIN
Public Function ProposeGlobalLineTrade(portfolios As Dictionary, selectedISIN As String) As Dictionary

    Dim globalTrades As New Dictionary ' Key: PortName -> Dictionary of Trades
    Dim pKey As Variant
    Dim port As Portfolio
    Dim bench As Portfolio
    Dim trades As Dictionary

    For Each pKey In portfolios
        Set port = portfolios(pKey)

        If Not port.benchmark Is Nothing Then
            If port.benchmark.Exists("composition") Then
                Set bench = port.benchmark("composition")

                ' Run the proposal logic for this specific portfolio and its benchmark
                Set trades = ProposeSingleLineTrade(port, bench, selectedISIN)

                If trades.count > 0 Then
                    globalTrades.Add port.Name, trades
                End If
            End If
        End If
    Next pKey

    Set ProposeGlobalLineTrade = globalTrades

End Function

' Finds the ISIN with the largest absolute weight discrepancy between port and bench (must be > 2%)
' and proposes the trades needed to fix it.
Public Function ProposeLargestDiscrepancyTrade(port As Portfolio, bench As Portfolio) As Dictionary

    Dim proposedTrades As New Dictionary

    If port.positions Is Nothing Or bench.positions Is Nothing Then
        Set ProposeLargestDiscrepancyTrade = proposedTrades
        Exit Function
    End If

    Dim uniqueISINs As New Dictionary
    Dim bKey As Variant, pKey As Variant
    Dim pos As Position
    Dim isCash As Boolean

    ' 1. Collect all unique non-cash ISINs
    For Each bKey In bench.positions
        Set pos = bench.positions(bKey)
        isCash = False
        If Not pos.posInstrument Is Nothing Then
            If LCase(Trim(pos.posInstrument.category)) = "cash" Then isCash = True
        ElseIf pos.posID = "10" Or pos.posID = pos.ccy Then
            isCash = True
        End If
        If pos.isin <> "" And Not isCash Then uniqueISINs(pos.isin) = 1
    Next bKey

    For Each pKey In port.positions
        Set pos = port.positions(pKey)
        isCash = False
        If Not pos.posInstrument Is Nothing Then
            If LCase(Trim(pos.posInstrument.category)) = "cash" Then isCash = True
        ElseIf pos.posID = "10" Or pos.posID = pos.ccy Then
            isCash = True
        End If
        If pos.isin <> "" And Not isCash Then uniqueISINs(pos.isin) = 1
    Next pKey

    ' 2. Evaluate Discrepancies
    Dim isin As Variant
    Dim maxDiff As Double: maxDiff = 0
    Dim targetISIN As String: targetISIN = ""

    Dim targetW As Double
    Dim currentW As Double
    Dim diff As Double

    For Each isin In uniqueISINs.Keys
        targetW = 0
        currentW = 0

        ' Get Target
        For Each bKey In bench.positions
            If bench.positions(bKey).isin = isin Then
                targetW = bench.positions(bKey).weight
                Exit For
            End If
        Next bKey

        ' Get Current
        For Each pKey In port.positions
            If port.positions(pKey).isin = isin Then
                currentW = port.positions(pKey).weight
                Exit For
            End If
        Next pKey

        diff = Abs(targetW - currentW)

        If diff > maxDiff Then
            maxDiff = diff
            targetISIN = CStr(isin)
        End If
    Next isin

    ' 3. Propose if > 2%
    If maxDiff >= 0.02 And targetISIN <> "" Then
        Set proposedTrades = ProposeSingleLineTrade(port, bench, targetISIN)
    End If

    Set ProposeLargestDiscrepancyTrade = proposedTrades

End Function

' Iterates through all portfolios, finds their individual largest discrepancy, and proposes trades.
Public Function ProposeAllLargestDiscrepancies(portfolios As Dictionary) As Dictionary

    Dim globalTrades As New Dictionary
    Dim pKey As Variant
    Dim port As Portfolio
    Dim bench As Portfolio
    Dim trades As Dictionary

    For Each pKey In portfolios
        Set port = portfolios(pKey)

        If Not port.benchmark Is Nothing Then
            If port.benchmark.Exists("composition") Then
                Set bench = port.benchmark("composition")

                Set trades = ProposeLargestDiscrepancyTrade(port, bench)

                If trades.count > 0 Then
                    globalTrades.Add port.Name, trades
                End If
            End If
        End If
    Next pKey

    Set ProposeAllLargestDiscrepancies = globalTrades

End Function

' Generates an HTML email summarizing the global trades
Public Sub EmailGlobalLineTrades(globalTrades As Dictionary, Optional contextString As String = "")

    If globalTrades.count = 0 Then
        MsgBox "No trades required across the portfolios based on current parameters.", vbInformation
        Exit Sub
    End If

    Dim htmlBody As String
    htmlBody = "<html><body style='font-family: Arial, sans-serif;'>"

    If contextString <> "" Then
        htmlBody = htmlBody & "<h2>Proposed Trades for: " & contextString & "</h2>"
    Else
        htmlBody = htmlBody & "<h2>Proposed Trades: Largest Discrepancies (>2%)</h2>"
    End If

    Dim portName As Variant
    Dim trades As Dictionary
    Dim tKey As Variant
    Dim tPos As Position

    htmlBody = htmlBody & "<table border='1' style='border-collapse: collapse; width: 90%;'>"
    htmlBody = htmlBody & "<tr style='background-color: #f2f2f2;'><th style='padding: 8px; text-align: left;'>Portfolio</th><th style='padding: 8px; text-align: left;'>Instrument Name</th><th style='padding: 8px; text-align: center;'>Action</th><th style='padding: 8px; text-align: right;'>Quantity</th><th style='padding: 8px; text-align: right;'>Amount (Base CCY)</th><th style='padding: 8px; text-align: left;'>Note</th></tr>"

    For Each portName In globalTrades
        Set trades = globalTrades(portName)

        For Each tKey In trades
            Set tPos = trades(tKey)

            Dim actionStr As String
            If tPos.qty > 0 Then actionStr = "BUY" Else actionStr = "SELL"

            Dim colorStr As String
            If actionStr = "BUY" Then colorStr = "green" Else colorStr = "red"

            Dim noteStr As String
            ' In the "largest discrepancy" mode, there's no single target ISIN,
            ' so we assume trades that end in "_FUNDING" are cash management trades.
            If Right(CStr(tKey), 8) = "_FUNDING" Then
                noteStr = "Funding Trade"
            Else
                noteStr = "Target Trade"
            End If

            htmlBody = htmlBody & "<tr>"
            htmlBody = htmlBody & "<td style='padding: 8px; font-weight: bold;'>" & portName & "</td>"
            htmlBody = htmlBody & "<td style='padding: 8px;'>" & tPos.name & "</td>"
            htmlBody = htmlBody & "<td style='padding: 8px; text-align: center; color: " & colorStr & "; font-weight: bold;'>" & actionStr & "</td>"
            htmlBody = htmlBody & "<td style='padding: 8px; text-align: right;'>" & Format(Abs(tPos.qty), "#,##0.00") & "</td>"
            htmlBody = htmlBody & "<td style='padding: 8px; text-align: right;'>" & Format(Abs(tPos.amountBaseCur), "#,##0") & " " & tPos.ccy & "</td>"
            htmlBody = htmlBody & "<td style='padding: 8px; font-style: italic; color: gray;'>" & noteStr & "</td>"
            htmlBody = htmlBody & "</tr>"

        Next tKey
    Next portName

    htmlBody = htmlBody & "</table>"
    htmlBody = htmlBody & "<br><p>Regards,<br>Trade Proposer System</p></body></html>"

    ' Generate Email via Outlook
    On Error Resume Next
    Dim outApp As Object
    Dim outMail As Object
    Set outApp = CreateObject("Outlook.Application")
    If outApp Is Nothing Then
        MsgBox "Cannot open Outlook to generate trade proposal email.", vbExclamation
        Exit Sub
    End If

    Set outMail = outApp.CreateItem(0) ' olMailItem
    With outMail
        If contextString <> "" Then
            .Subject = "Trade Proposals: " & contextString & " - " & Format(Date, "YYYY-MM-DD")
        Else
            .Subject = "Trade Proposals: Largest Discrepancies - " & Format(Date, "YYYY-MM-DD")
        End If
        .htmlBody = htmlBody
        .Display ' Show the draft
    End With
    On Error GoTo 0

End Sub
