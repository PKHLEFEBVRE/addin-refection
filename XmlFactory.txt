
Private Function LoadXmlDoc(xmlPath As String) As MSXML2.DOMDocument60

    Dim doc As New MSXML2.DOMDocument60
    
    doc.async = False ' one time load
    
    If Not doc.Load(xmlPath) Then
        Err.Raise vbObjectError + 1000, , "Failed to load XML: " & doc.parseError.reason
    End If
    
    Set LoadXmlDoc = doc
    
End Function


Private Function BuildSecbaseLookup(xmlDoc As MSXML2.DOMDocument60) As Dictionary

    Dim lookup As New Dictionary
    Dim node   As MSXML2.IXMLDOMNode
    
    For Each node In xmlDoc.SelectNodes("//Secbase")
        lookup(node.SelectSingleNode("Isin").Text) = node.SelectSingleNode("SecName").Text
    Next
    
    Set BuildSecbaseLookup = lookup
    
End Function


Private Sub AddSecurityPositions(xmlDoc As MSXML2.DOMDocument60, positions As Dictionary, names As Dictionary, ByVal mandate As String, ByRef instruments As Dictionary)

    Dim spNode As MSXML2.IXMLDOMNode
    Dim pos As Position
    
    Dim config As New Configuration
    Dim assetClass As Variant, posInstrument As Instrument, cashPosition As Position
    Dim aum As Double, tick As Variant, isin As Variant
    
    aum = 0
    For Each posID In positions
        aum = aum + positions(posID).amountBaseCur
    Next posID
    
    For Each spNode In xmlDoc.SelectNodes("//Secpos")
        
        Set posInstrument = Nothing
        Set cashPosition = Nothing
        
        If spNode.ParentNode.ParentNode.Attributes.getNamedItem("ContId").Text = mandate Then
        
            For Each assetClass In config.validInstrumentsAssetClassesForBDLReferential
                If instruments.Exists(assetClass) Then
                    For Each tick In instruments(assetClass)
                        For Each isin In instruments(assetClass)(tick)
                            If isin = spNode.SelectSingleNode("Isin").Text And spNode.SelectSingleNode("PriceCcy").Text = instruments(assetClass)(tick)(isin).ccy Then
                                Set posInstrument = instruments(assetClass)(tick)(isin)
                                Exit For
                            End If
                        Next isin
                        If Not posInstrument Is Nothing Then Exit For
                    Next tick
                End If
                If Not posInstrument Is Nothing Then Exit For
            Next assetClass
            
            If posInstrument Is Nothing Then
                MsgBox ("Mandat " & mandate & "Unable to find " & spNode.SelectSingleNode("Isin").Text & _
    " in the referential, please create it in Infin and replace the referential by the new one in the inventory folder.")
                Exit Sub
            End If
            
            
            For Each tick In positions
                If LCase(positions(tick).ccy) = LCase(spNode.SelectSingleNode("PriceCcy").Text) Then
                    Set cashPosition = positions(tick)
                    Exit For
                End If
            Next tick
            
            If cashPosition Is Nothing Then
                Dim manualFx As Double
                manualFx = 1#
                If spNode.SelectSingleNode("PriceCcy").Text = "BRL" Then manualFx = 0.16
                MsgBox ("Mandat " & mandate & "Unable to find the " & spNode.SelectSingleNode("PriceCcy").Text & " in the cash accounts, setting it to 0 and FX rate at " & manualFx)
                Set cashPosition = PositionFactory.CreatePosition( _
                    "", _
                    spNode.ParentNode.ParentNode.Attributes.getNamedItem("ContId").Text, _
                    spNode.ParentNode.ParentNode.Attributes.getNamedItem("ContId").Text, _
                    spNode.SelectSingleNode("AssBalDate").Text, _
                    spNode.SelectSingleNode("PriceCcy").Text, _
                    "", _
                    "", _
                    0#, _
                    manualFx, _
                    spNode.SelectSingleNode("PriceCcy").Text, _
                    manualFx, _
                    Nothing _
                )
            End If
            
            
            Set pos = PositionFactory.CreatePosition( _
                posInstrument.instrumentId, _
                spNode.ParentNode.ParentNode.Attributes.getNamedItem("ContId").Text, _
                spNode.ParentNode.ParentNode.Attributes.getNamedItem("ContId").Text, _
                spNode.SelectSingleNode("AssBalDate").Text, _
                posInstrument.name, _
                posInstrument.isin, _
                posInstrument.ticker, _
                spNode.SelectSingleNode("AssetQty").Text, _
                spNode.SelectSingleNode("UnitPrice").Text, _
                spNode.SelectSingleNode("PriceCcy").Text, _
                cashPosition.fx, _
                posInstrument _
            ) 'name was > IIf(names.Exists(spNode.SelectSingleNode("Isin").Text), names(spNode.SelectSingleNode("Isin").Text), ""), _

            positions.Add pos.posID, pos
            aum = aum + pos.amountBaseCur
            
        End If
        
    Next spNode
    
    If aum > 0 And positions.count > 0 Then
        For Each posID In positions
            positions(posID).weight = positions(posID).amountBaseCur / aum
        Next posID
    End If
    
End Sub



Private Sub AddCashPositions(xmlDoc As MSXML2.DOMDocument60, positions As Dictionary, ByVal mandate As String, ByRef instruments As Dictionary)
    
    Dim pos As Position, fx As Double, posInstrument As Instrument
    Dim contEl As MSXML2.IXMLDOMElement
    Dim cpNode As MSXML2.IXMLDOMNode
    
    For Each cpNode In xmlDoc.SelectNodes("//Cashpos")
        
        Set posInstrument = Nothing
        
        If cpNode.ParentNode.ParentNode.Attributes.getNamedItem("ContId").Text = mandate Then
        
            If instruments("cash").Exists(cpNode.SelectSingleNode("AssetCcy").Text) Then _
                Set posInstrument = instruments("cash")(cpNode.SelectSingleNode("AssetCcy").Text)(cpNode.SelectSingleNode("AssetCcy").Text)
            
            fx = 1#
            If CDbl(cpNode.SelectSingleNode("balBook").Text) <> 0 Then fx = CDbl(cpNode.SelectSingleNode("PosBalRefCcy").Text) / CDbl(cpNode.SelectSingleNode("balBook").Text)
            
            
            
            Set pos = PositionFactory.CreatePosition( _
                posInstrument.instrumentId, _
                cpNode.ParentNode.ParentNode.Attributes.getNamedItem("ContId").Text, _
                cpNode.ParentNode.ParentNode.Attributes.getNamedItem("ContId").Text, _
                cpNode.SelectSingleNode("PosBalDate").Text, _
                cpNode.SelectSingleNode("AssetCcy").Text, _
                "", _
                "", _
                cpNode.SelectSingleNode("balBook").Text, _
                1#, _
                cpNode.SelectSingleNode("AssetCcy").Text, _
                fx, _
                posInstrument _
            )
            
            'If pos.qty <> 0 Then positions.Add pos.posID, pos
            positions.Add pos.posID, pos
            
        End If
        
    Next cpNode
    
End Sub



Public Function CreatePositionsDictFromXmlPath(path As String, ByVal mandateId As String, ByRef instruments As Dictionary) As Dictionary

    Dim xmlDoc As MSXML2.DOMDocument60
    Dim positions  As New Dictionary
    Dim secbaseDict As Dictionary

    Set xmlDoc = LoadXmlDoc(path)
    Set secbaseDict = BuildSecbaseLookup(xmlDoc)

    AddCashPositions xmlDoc, positions, mandateId, instruments
    AddSecurityPositions xmlDoc, positions, secbaseDict, mandateId, instruments
    
    Dim pos As Variant, aum As Double
    aum = 0
    
    For Each pos In positions
        aum = aum + positions(pos).amountBaseCur
    Next pos
    
    If aum > 0 And positions.count > 0 Then
        Set CreatePositionsDictFromXmlPath = positions
    End If
    
End Function
