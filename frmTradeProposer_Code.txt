Option Explicit

' --- Instructions for User ---
' 1. Create a UserForm in the VBA Editor (Insert -> UserForm).
' 2. Name it "frmTradeProposer".
' 3. Add a ListBox named "lstBenchmarkISINs". Set ColumnCount to 2 (ISIN, Name).
' 4. Add a CommandButton named "btnCalculate".
' 5. Add a TextBox named "txtResult" (Set MultiLine = True, ScrollBars = fmScrollBarsVertical).
' 6. Paste this code into the UserForm's code module.
' -----------------------------

Private portfolios As Dictionary

Private Sub UserForm_Initialize()
    ' 1. Load Portfolios
    ' Assuming data.GetPortfolios(Date) gives the dictionary
    Set portfolios = Data.GetPortfolios(Date)

    ' 2. Build unique list of ISINs from all benchmarks
    lstBenchmarkISINs.Clear
    Dim uniqueISINs As New Dictionary

    Dim pKey As Variant
    Dim port As Portfolio
    Dim bench As Portfolio
    Dim bKey As Variant
    Dim pos As Position
    Dim isCash As Boolean

    For Each pKey In portfolios
        Set port = portfolios(pKey)

        If Not port.benchmark Is Nothing Then
            If port.benchmark.Exists("composition") Then
                Set bench = port.benchmark("composition")

                If Not bench.positions Is Nothing Then
                    For Each bKey In bench.positions
                        Set pos = bench.positions(bKey)

                        ' Check if cash
                        isCash = False
                        If Not pos.posInstrument Is Nothing Then
                            If LCase(Trim(pos.posInstrument.category)) = "cash" Then isCash = True
                        ElseIf pos.posID = "10" Or pos.posID = pos.ccy Then
                            isCash = True
                        End If

                        If pos.isin <> "" And Not isCash Then
                            If Not uniqueISINs.Exists(pos.isin) Then
                                uniqueISINs.Add pos.isin, pos.name
                            End If
                        End If
                    Next bKey
                End If
            End If
        End If
    Next pKey

    ' Populate ListBox
    Dim uKey As Variant
    For Each uKey In uniqueISINs.Keys
        lstBenchmarkISINs.AddItem uKey
        lstBenchmarkISINs.List(lstBenchmarkISINs.ListCount - 1, 1) = uniqueISINs(uKey)
    Next uKey

End Sub

Private Sub btnCalculate_Click()

    If lstBenchmarkISINs.ListIndex = -1 Then
        MsgBox "Please select an ISIN from the benchmark list.", vbExclamation
        Exit Sub
    End If

    Dim selectedISIN As String
    selectedISIN = lstBenchmarkISINs.List(lstBenchmarkISINs.ListIndex, 0)

    ' Call the backend function
    Dim globalTrades As Dictionary
    Set globalTrades = TradeProposer.ProposeGlobalLineTrade(portfolios, selectedISIN)

    ' Display Results locally in the form
    If globalTrades.count = 0 Then
        txtResult.Text = "No trades needed. Target matches current weight across all applicable portfolios."
        Exit Sub
    End If

    Dim resultText As String
    resultText = "PROPOSED TRADES (Also emailed):" & vbCrLf & String(40, "-") & vbCrLf

    Dim portName As Variant
    Dim tKey As Variant
    Dim tPos As Position

    For Each portName In globalTrades
        resultText = resultText & "PORTFOLIO: " & portName & vbCrLf

        For Each tKey In globalTrades(portName)
            Set tPos = globalTrades(portName)(tKey)
            resultText = resultText & "  ISIN: " & tPos.isin & " (" & tPos.name & ")" & vbCrLf
            resultText = resultText & "  Action: " & IIf(tPos.qty > 0, "BUY", "SELL") & vbCrLf
            resultText = resultText & "  Qty: " & Format(tPos.qty, "#,##0.00") & vbCrLf
            resultText = resultText & "  Amt: " & Format(tPos.amountBaseCur, "#,##0") & " " & tPos.ccy & vbCrLf
            resultText = resultText & vbCrLf
        Next tKey

        resultText = resultText & String(30, "-") & vbCrLf
    Next portName

    txtResult.Text = resultText

    ' Generate the HTML Email
    TradeProposer.EmailGlobalLineTrades globalTrades, selectedISIN

End Sub
