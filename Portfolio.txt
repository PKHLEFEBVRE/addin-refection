Public comment As String

Public insuranceContract As String
Public accountName As String

Public insurer As String
Public custodian As String
Public sales As String

Public AVType As String
Public ccy As String
Public profile As String
Public composition As String
Public targetProfile As String
Public targetComposition As String
Public lastComputed As String
Public CashAndTreso As Double
Public AUM As Double
Public exposureLimits As Dictionary
Public positions As Dictionary

Private m_equityDelta As Double
Private m_equityDeltaComputed As Boolean

Private Sub Class_Initialize()

    Set exposureLimits = New Dictionary
    Set positions = New Dictionary
    m_equityDeltaComputed = False

End Sub

Public Function GetPositions(dt As Date) As Dictionary

    ' If positions are already loaded, return them
    If positions.count > 0 Then
        Set GetPositions = positions
        Exit Function
    End If

    ' Otherwise, load them
    Dim inventoryId As String
    inventoryId = InfinId
    If inventoryId = "" Then inventoryId = Name

    If inventoryId <> "" Then
        Set positions = Data.GetInventory(dt, Me)

        ' Compute AUM and Weights
        Dim key As Variant
        Dim pos As Position
        Dim totalAmount As Double

        totalAmount = 0
        For Each key In positions
            Set pos = positions(key)
            totalAmount = totalAmount + pos.amountBaseCur
        Next key

        Me.AUM = totalAmount

        If Me.AUM <> 0 Then
            For Each key In positions
                Set pos = positions(key)
                pos.weight = pos.amountBaseCur / Me.AUM
            Next key
        End If

    End If

    ' Reset computed values when positions change (if GetPositions is called)
    m_equityDeltaComputed = False

    Set GetPositions = positions

End Function

Public Function GetComputedEquityDelta() As Double

    If m_equityDeltaComputed Then
        GetComputedEquityDelta = m_equityDelta
        Exit Function
    End If

    Dim sumDelta As Double
    Dim pos As Variant
    Dim currentPos As Position

    sumDelta = 0

    If Not positions Is Nothing Then
        For Each pos In positions.Items
            Set currentPos = pos
            If Not currentPos.posInstrument Is Nothing Then
                sumDelta = sumDelta + (currentPos.weight * currentPos.posInstrument.equityDelta)
            End If
        Next pos
    End If

    m_equityDelta = sumDelta
    m_equityDeltaComputed = True

    GetComputedEquityDelta = m_equityDelta

End Function
