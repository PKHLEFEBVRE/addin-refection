Private m_equityDelta As Double
Private m_equityDeltaComputed As Boolean

Private Sub Class_Initialize()

    Set exposureLimits = New Dictionary
    Set positions = New Dictionary
    Set trades = New Dictionary
    m_equityDeltaComputed = False

End Sub

Public Function GetPositions(dt As Date) As Dictionary

    ' If positions are already loaded, return them
    If positions.count > 0 Then
        Set GetPositions = positions
        Exit Function
    End If

    ' Otherwise, load them
    Dim inventoryId As String
    inventoryId = InfinId
    If inventoryId = "" Then inventoryId = Name

    If inventoryId <> "" Then
        Set positions = Data.GetInventory(dt, Me)

        ' Compute AUM and Weights
        Dim key As Variant
        Dim pos As Position
        Dim totalAmount As Double

        totalAmount = 0
        For Each key In positions
            Set pos = positions(key)
            totalAmount = totalAmount + pos.amountBaseCur
        Next key

        Me.AUM = totalAmount

        If Me.AUM <> 0 Then
            For Each key In positions
                Set pos = positions(key)
                pos.weight = pos.amountBaseCur / Me.AUM
            Next key
        End If

    End If

    ' Reset computed values when positions change (if GetPositions is called)
    m_equityDeltaComputed = False

    Set GetPositions = positions

End Function

Public Sub SetTrades(tradesDict As Dictionary)
    If Not tradesDict Is Nothing Then
        Set trades = tradesDict
    End If
End Sub

Public Function GetPositionsWithTrades() As Dictionary

    Dim simPositions As Dictionary
    Set simPositions = New Dictionary

    Dim key As Variant
    Dim pos As Position, newPos As Position
    Dim trade As Position
    Dim totalTradeCost As Double

    ' 1. Clone original positions
    For Each key In positions
        Set pos = positions(key)
        Set newPos = pos.Clone()
        simPositions.Add newPos.posID, newPos
    Next key

    ' 2. Calculate trade cost (excluding Cash "10")
    totalTradeCost = 0
    For Each key In trades
        Set trade = trades(key)
        If trade.posID <> "10" Then
            totalTradeCost = totalTradeCost + trade.amountBaseCur
        End If
    Next key

    ' 3. Apply trades
    For Each key In trades
        Set trade = trades(key)

        If simPositions.Exists(trade.posID) Then
            Set newPos = simPositions(trade.posID)
            newPos.qty = newPos.qty + trade.qty
            newPos.amountBaseCur = newPos.amountBaseCur + trade.amountBaseCur
        Else
            Set newPos = trade.Clone()
            simPositions.Add newPos.posID, newPos
        End If
    Next key

    ' 4. Automatic Financing (Subtract cost from Cash "10")
    If simPositions.Exists("10") Then
        Set newPos = simPositions("10")
        newPos.amountBaseCur = newPos.amountBaseCur - totalTradeCost
    Else
        ' Should not happen based on user input, but good to handle or log
        Debug.Print "Warning: Cash position '10' not found in portfolio " & Name
    End If

    ' 5. Recompute Weights
    Dim simAUM As Double
    simAUM = 0

    For Each key In simPositions
        Set newPos = simPositions(key)
        simAUM = simAUM + newPos.amountBaseCur
    Next key

    If simAUM <> 0 Then
        For Each key In simPositions
            Set newPos = simPositions(key)
            newPos.weight = newPos.amountBaseCur / simAUM
        Next key
    End If

    Set GetPositionsWithTrades = simPositions
Public Function GetComputedEquityDelta() As Double

    If m_equityDeltaComputed Then
        GetComputedEquityDelta = m_equityDelta
        Exit Function
    End If

    Dim sumDelta As Double
    Dim pos As Variant
    Dim currentPos As Position

    sumDelta = 0

    If Not positions Is Nothing Then
        For Each pos In positions.Items
            Set currentPos = pos
            If Not currentPos.posInstrument Is Nothing Then
                sumDelta = sumDelta + (currentPos.weight * currentPos.posInstrument.equityDelta)
            End If
        Next pos
    End If

    m_equityDelta = sumDelta
    m_equityDeltaComputed = True

    GetComputedEquityDelta = m_equityDelta

End Function
